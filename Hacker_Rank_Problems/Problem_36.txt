    public static boolean solve(int index, int leap, int[] game, boolean[] visited) {

        // If we go outside the array, we win
        if (index >= game.length) {
            return true;
        }

        // Invalid position
        if (index < 0) {
            return false;
        }

        // Cannot move to a cell containing 1
        if (game[index] == 1) {
            return false;
        }

        // Already visited this position
        if (visited[index]) {
            return false;
        }

        // Mark as visited
        visited[index] = true;

        // Try moving forward by 1
        if (solve(index + 1, leap, game, visited)) {
            return true;
        }

        // Try jumping by leap
        if (solve(index + leap, leap, game, visited)) {
            return true;
        }

        // Try moving backward by 1
        if (solve(index - 1, leap, game, visited)) {
            return true;
        }

        return false;
    }
